#!/bin/bash

# Zlyme PortMaster launch environment. Every port .sh sources this as
# $controlfolder/control.txt. Do not source /etc/profile: that file sets
# HOME=/storage and would dump ~/.local onto the OS card.
#
# HOME / XDG_* stay on the OS card (nextui-session). PORTS.pak bind-mounts
# /roms/ports onto that volume's Ports folder for the launch.
# /opt/system/Tools/PortMaster is the selected library's live tree.

CUR_TTY=/dev/tty0

export directory="${directory:-roms}"

if [ -z "${controlfolder:-}" ] || [ ! -d "$controlfolder" ]; then
	if [ -d /opt/system/Tools/PortMaster ]; then
		controlfolder=/opt/system/Tools/PortMaster
	elif [ -x /usr/sbin/zlyme-portmaster-prepare ]; then
		eval "$(/usr/sbin/zlyme-portmaster-prepare export)" || exit 1
		controlfolder="${HM_TOOLS_DIR}/PortMaster"
	fi
fi
export controlfolder

# Same library zlyme-portmaster-root selected. Do not invent a fallback disk.
if [ -z "${HM_TOOLS_DIR:-}" ] || [ -z "${HM_PORTS_DIR:-}" ]; then
	if [ -x /usr/sbin/zlyme-portmaster-prepare ]; then
		eval "$(/usr/sbin/zlyme-portmaster-prepare export)" || exit 1
	else
		echo "PortMaster is not ready" >&2
		exit 1
	fi
fi
export HM_TOOLS_DIR
if [ -z "${HM_SCRIPTS_DIR:-}" ]; then
	HM_SCRIPTS_DIR=$HM_PORTS_DIR
fi
export HM_PORTS_DIR HM_SCRIPTS_DIR
mkdir -p "$HM_PORTS_DIR" "$HM_TOOLS_DIR/PortMaster/libs" "$HM_TOOLS_DIR/PortMaster/runtimes"

# Prefix only. ESUDOKILL / ESUDOKILL2 stay the gptokeyb signal arguments.
ESUDO=/usr/sbin/zlyme-portmaster-exec
ESUDOKILL="-1"
ESUDOKILL2="-1"
export ESUDO ESUDOKILL ESUDOKILL2
export PM_CAN_MOUNT="${PM_CAN_MOUNT:-Y}"

export SDL_VIDEODRIVER="${SDL_VIDEODRIVER:-kmsdrm}"
# Same boundary as pak-input.sh. A port that re-sources this file
# after clearing its environment still sees only xb360 targets.
export SDL_GAMECONTROLLER_IGNORE_DEVICES_EXCEPT="${SDL_GAMECONTROLLER_IGNORE_DEVICES_EXCEPT:-0x045e/0x028e}"
if [ -z "${SDL_GAMECONTROLLERCONFIG_FILE:-}" ] || [ ! -f "${SDL_GAMECONTROLLERCONFIG_FILE:-}" ]; then
	if [ -f /usr/lib/gamecontrollerdb.txt ]; then
		export SDL_GAMECONTROLLERCONFIG_FILE=/usr/lib/gamecontrollerdb.txt
	elif [ -f "$controlfolder/gamecontrollerdb.txt" ]; then
		export SDL_GAMECONTROLLERCONFIG_FILE="$controlfolder/gamecontrollerdb.txt"
	fi
fi

raloc="/usr/bin"
raconf="--config /storage/.config/retroarch/retroarch.cfg"

[ -f "$controlfolder/device_info.txt" ] && . "$controlfolder/device_info.txt"
[ -f "$controlfolder/funcs.txt" ] && . "$controlfolder/funcs.txt"

# device_info.txt assigns CFW_NAME=Unknown and may replace
# DEVICE_NAME from the devicetree. Ports then look for mod_${CFW_NAME}.
export CFW_NAME=Zlyme
export DEVICE_NAME="Miyoo Flip"
: "${DISPLAY_WIDTH:=640}"
: "${DISPLAY_HEIGHT:=480}"
export DISPLAY_WIDTH DISPLAY_HEIGHT

get_controls() {
	: "${SDL_GAMECONTROLLERCONFIG_FILE:=/usr/lib/gamecontrollerdb.txt}"
	if [ ! -f "$SDL_GAMECONTROLLERCONFIG_FILE" ] && [ -f "$controlfolder/gamecontrollerdb.txt" ]; then
		SDL_GAMECONTROLLERCONFIG_FILE="$controlfolder/gamecontrollerdb.txt"
	fi
	export SDL_GAMECONTROLLERCONFIG_FILE
	export SDL_GAMECONTROLLER_IGNORE_DEVICES_EXCEPT="${SDL_GAMECONTROLLER_IGNORE_DEVICES_EXCEPT:-0x045e/0x028e}"
	if [ -f "$SDL_GAMECONTROLLERCONFIG_FILE" ]; then
		cp -f "$SDL_GAMECONTROLLERCONFIG_FILE" /tmp/gamecontrollerdb.txt 2>/dev/null || true
		if [ -z "${SDL_GAMECONTROLLERCONFIG:-}" ]; then
			SDL_GAMECONTROLLERCONFIG=$(grep -v '^#' "$SDL_GAMECONTROLLERCONFIG_FILE" | grep -v '^$')
			export SDL_GAMECONTROLLERCONFIG
		fi
		sdl_controllerconfig="${SDL_GAMECONTROLLERCONFIG:-$(cat "$SDL_GAMECONTROLLERCONFIG_FILE")}"
	else
		sdl_controllerconfig="${SDL_GAMECONTROLLERCONFIG:-}"
	fi
	: "${ANALOG_STICKS:=2}"
	: "${LOWRES:=N}"
	: "${param_device:=rg552}"
}

export GPTOKEYB2="$ESUDO env LD_PRELOAD=$controlfolder/libinterpose.${DEVICE_ARCH:-aarch64}.so $controlfolder/gptokeyb2 $ESUDOKILL2"
export GPTOKEYB="$ESUDO $controlfolder/gptokeyb $ESUDOKILL"
